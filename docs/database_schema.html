Human-In-The-Loop (HITL) Audit Architecture

Every time a warehouse manager approves an allocation, overrides a Tier 3 substitute, or modifies a planogram target, the change is immutably recorded in the audit_logs table using PostgreSQL's native JSONB datatype.

Audit Log Table Schema

CREATE TABLE audit_logs (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    entity_name VARCHAR(100) NOT NULL, -- e.g. 'allocation_dispatches'
    entity_id UUID NOT NULL,
    action_type VARCHAR(50) NOT NULL, -- 'APPROVE', 'OVERRIDE', 'REJECT'
    changed_by_user VARCHAR(255) NOT NULL,
    old_values JSONB, -- Previous state snapshot
    new_values JSONB, -- Updated state snapshot
    override_reason TEXT, -- Manager justification
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

Automatic PostgreSQL Trigger

CREATE OR REPLACE FUNCTION log_dispatch_override()
RETURNS TRIGGER AS $$
BEGIN
  IF NEW.status <> OLD.status THEN
    INSERT INTO audit_logs (
      entity_name, entity_id, action_type, 
      changed_by_user, old_values, new_values
    )
    VALUES (
      'allocation_dispatches', NEW.id, NEW.status,
      CURRENT_USER, to_jsonb(OLD), to_jsonb(NEW)
    );
  END IF;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;
